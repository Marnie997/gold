nthSundayOfMonth(_year, _month, _nth, _tz) =>
    first_ts  = timestamp(_tz, _year, _month, 1, 0, 0)
    first_dow = dayofweek(first_ts)
    offset    = (dayofweek.sunday - first_dow + 7) % 7
    1 + offset + (_nth - 1) * 7

lastSundayOfMonth(_year, _month, _tz) =>
    next_month_ts = _month == 12 ? timestamp(_tz, _year + 1, 1, 1, 0, 0) : timestamp(_tz, _year, _month + 1, 1, 1, 0, 0)
    last_day_ts   = next_month_ts - 24 * 60 * 60 * 1000
    last_day_num  = dayofmonth(last_day_ts, _tz)
    last_dow      = dayofweek(last_day_ts)
    offset        = (last_dow - dayofweek.sunday + 7) % 7
    last_day_num - offset

isUSDST() =>
    y         = year(time, "America/New_York")
    start_day = nthSundayOfMonth(y, 3, 2, "America/New_York")
    end_day   = nthSundayOfMonth(y, 11, 1, "America/New_York")
    start_ts  = timestamp("America/New_York", y, 3, start_day, 2, 0)
    end_ts    = timestamp("America/New_York", y, 11, end_day, 2, 0)
    time >= start_ts and time < end_ts

isEuroDST() =>
    y         = year(time, "Europe/Berlin")
    start_day = lastSundayOfMonth(y, 3, "Europe/Berlin")
    end_day   = lastSundayOfMonth(y, 10, "Europe/Berlin")
    start_ts  = timestamp("Europe/Berlin", y, 3, start_day, 2, 0)
    end_ts    = timestamp("Europe/Berlin", y, 10, end_day, 3, 0)
    time >= start_ts and time < end_ts

sessionStartTs(_h, _m) =>
    timestamp("GMT+9", year(time, "GMT+9"), month(time, "GMT+9"), dayofmonth(time, "GMT+9"), _h, _m)

isFirstBarAfterSession(_h, _m) =>
    sess = sessionStartTs(_h, _m)
    time >= sess and (na(time[1]) or time[1] < sess)

inAsiaSession() =>
    curMin    = hour(time, "GMT+9") * 60 + minute(time, "GMT+9")
    asiaStart = asia_h * 60 + asia_m
    euroStart = (euro_dst ? 16 : 17) * 60
    curMin >= asiaStart and curMin < euroStart

inEuroSession() =>
    curMin    = hour(time, "GMT+9") * 60 + minute(time, "GMT+9")
    euroStart = (euro_dst ? 16 : 17) * 60
    usStart   = (us_dst ? 22 : 23) * 60 + 30
    curMin >= euroStart and curMin < usStart

inUSSession() =>
    curMin  = hour(time, "GMT+9") * 60 + minute(time, "GMT+9")
    usStart = (us_dst ? 22 : 23) * 60 + 30
    curMin >= usStart or curMin < (asia_h * 60 + asia_m)

combineTwoBarRange(_prevHigh, _prevLow, _currHigh, _currLow) =>
    float out = na
    if na(_prevHigh) or na(_prevLow) or na(_currHigh) or na(_currLow)
        out := na
    else
        out := math.max(_prevHigh, _currHigh) - math.min(_prevLow, _currLow)
    out

getAsiaComposite(_tf, _sessionH, _sessionM) =>
    request.security(
         syminfo.tickerid,
         _tf,
         ta.valuewhen(
             barstate.isconfirmed and isFirstBarAfterSession(_sessionH, _sessionM) and not na(high[1]) and not na(low[1]),
             combineTwoBarRange(high[1], low[1], high, low),
             0
         ),
         gaps = barmerge.gaps_off,
         lookahead = barmerge.lookahead_off
     )

getOpenBarRange(_tf, _sessionH, _sessionM) =>
    request.security(
         syminfo.tickerid,
         _tf,
         ta.valuewhen(
             barstate.isconfirmed and isFirstBarAfterSession(_sessionH, _sessionM),
             high - low,
             0
         ),
         gaps = barmerge.gaps_off,
         lookahead = barmerge.lookahead_off
     )

calcRiskSplitQtys(_riskUsd, _setupKtr, _pointValue, _maxAddStep, _selectedStopKtr) =>
    float _q1 = na
    float _q2 = na
    float _q3 = na
    float _q4 = na
    float _q5 = na
    float _q6 = na

    bool _ready = not na(_riskUsd) and _riskUsd > 0 and not na(_setupKtr) and _setupKtr > 0 and _pointValue > 0
    if _ready
        float _totalSteps  = _maxAddStep + 1.0
        float _perStepRisk = _riskUsd / _totalSteps

        _q1 := _perStepRisk / (_selectedStopKtr * _setupKtr * _pointValue)
        _q2 := _maxAddStep >= 1 ? _perStepRisk / ((_selectedStopKtr - 1.0) * _setupKtr * _pointValue) : na
        _q3 := _maxAddStep >= 2 ? _perStepRisk / ((_selectedStopKtr - 2.0) * _setupKtr * _pointValue) : na
        _q4 := _maxAddStep >= 3 ? _perStepRisk / ((_selectedStopKtr - 3.0) * _setupKtr * _pointValue) : na
        _q5 := _maxAddStep >= 4 ? _perStepRisk / ((_selectedStopKtr - 4.0) * _setupKtr * _pointValue) : na
        _q6 := _maxAddStep >= 5 ? _perStepRisk / ((_selectedStopKtr - 5.0) * _setupKtr * _pointValue) : na

    [_q1, _q2, _q3, _q4, _q5, _q6]
